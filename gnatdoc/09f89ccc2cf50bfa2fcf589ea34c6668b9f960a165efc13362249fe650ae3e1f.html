ADC_Channel_Count

ADC_Channel_Count : constant :=
   (case Device is
      when RP2040             => 5,
      when RP2350A | RP2354A  => 5,
      when RP2350B | RP2354B  => 9);

ADC_Temperature_Channel

ADC_Temperature_Channel : constant :=
   (if Pin_Count = 48 then 8 else 4);

      when RP2350B | RP2354B  => 48);
